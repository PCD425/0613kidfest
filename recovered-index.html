<script>

  const COLLECTOR_URL='https://script.google.com/macros/s/AKfycbyXzQxI9z5hUbe1MzkoB6vWZw2w-HCyobWFmUTMTthEkcgbnrbOOO63k38G5lVkvTQcKA/exec';

  const responses=JSON.parse(localStorage.getItem('kidovaKidsfestResponses')||'[]'); const pilotLeads=JSON.parse(localStorage.getItem('kidovaPilotLeads')||'[]'); const pending=JSON.parse(localStorage.getItem('kidovaPendingSubmissions')||'[]'); const scaleValues={};

  const surveyForm=document.getElementById('form'),pilotForm=document.getElementById('pilotForm');

  function show(id){document.querySelectorAll('.screen').forEach(x=>x.classList.remove('active'));document.getElementById(id).classList.add('active');scrollTo(0,0)}

  function openExternal(url){window.open(url,'_blank','noopener,noreferrer')}

  function openDemo(){show('explore')}

  const kidFeedbackCopy={

    rush:'<strong>That could put me in more danger.</strong><br>If I rush ahead alone, I might get farther from the route I know, miss a crossing signal, or become harder for my grown-up to find. I should stop and choose a safer first step.',

    crowd:'<strong>A crowd may look like it knows the way, but it may be going somewhere else.</strong><br>Following it could lead me to the wrong exit, and I should never let an unknown person lead me away. I should pause where I am safe and easy to see.',

    safe:'<strong>Great safety choice!</strong><br>Pausing somewhere safe and visible gives me time to look around, check with Safe Route Buddy, and contact a trusted grown-up.'

  };

  async function kidChoice(el,choice){

    const ok=choice==='safe';

    document.querySelectorAll('#kids .choice').forEach(x=>x.disabled=true);el.classList.add(ok?'good':'bad');

    kidFeedback.classList.remove('hidden','good-feedback','risk-feedback');kidFeedback.classList.add(ok?'good-feedback':'risk-feedback');kidFeedback.innerHTML=kidFeedbackCopy[choice];

    kidFinish.classList.toggle('hidden',!ok);kidRetry.classList.toggle('hidden',ok);

    const data={submission_id:submissionId(),timestamp:new Date().toISOString(),selected_choice:el.textContent.trim(),choice_key:choice,choice_safe:String(ok),completed:String(ok)};

    await sendOrQueue({type:'kids_activity',data});

  }

  function retryKidChoice(){

    document.querySelectorAll('#kids .choice').forEach(x=>{x.disabled=false;x.classList.remove('good','bad')});

    kidFeedback.classList.add('hidden');kidRetry.classList.add('hidden');kidFinish.classList.add('hidden');

  }

  function pick(el,name){el.parentElement.querySelectorAll('button').forEach(x=>x.classList.remove('selected'));el.classList.add('selected');scaleValues[name]=el.textContent;progress()}

  function progress(){let f=surveyForm,filled=[...f.querySelectorAll('select')].filter(x=>x.value).length+Object.keys(scaleValues).length+(f.trust.value?1:0),total=f.querySelectorAll('select').length+2;bar.style.width=Math.min(100,filled/total*100)+'%'}

  function submissionId(){return crypto.randomUUID?crypto.randomUUID():'kidova-'+Date.now()+'-'+Math.random().toString(36).slice(2)}

  function savePending(){localStorage.setItem('kidovaPendingSubmissions',JSON.stringify(pending))}

  async function sendSubmission(item){

    const body=new URLSearchParams({payload:JSON.stringify(item)});

    await fetch(COLLECTOR_URL,{method:'POST',mode:'no-cors',body});

  }

  async function sendOrQueue(item){

    try{await sendSubmission(item);return true}

    catch(error){pending.push(item);savePending();return false}

  }

  async function retryPending(){

    if(!pending.length||!navigator.onLine)return;

    const retry=[...pending];pending.length=0;

    for(const item of retry){try{await sendSubmission(item)}catch(error){pending.push(item)}}

    savePending();

  }

  surveyForm.addEventListener('submit',async e=>{

    e.preventDefault();if(!scaleValues.current_solution_score)return alert('Please rate how well the current solution worked.');

    surveySubmit.disabled=true;surveySubmit.textContent='Submitting...';

    let data=Object.fromEntries(new FormData(surveyForm));Object.assign(data,scaleValues);data.submission_id=submissionId();data.timestamp=new Date().toISOString();

    responses.push(data);localStorage.setItem('kidovaKidsfestResponses',JSON.stringify(responses));

    const sent=await sendOrQueue({type:'survey',data});

    submissionStatus.innerHTML=sent?'<b>Response submitted.</b> Your anonymous answers were sent to the Kidova research sheet.':'<b>Saved on this device.</b> The response will retry automatically when the internet connection returns.';

    surveyForm.reset();Object.keys(scaleValues).forEach(k=>delete scaleValues[k]);bar.style.width='0';surveySubmit.disabled=false;surveySubmit.textContent='Finish survey';show('thanks');

  });

  pilotForm.addEventListener('submit',async e=>{

    e.preventDefault();pilotSubmit.disabled=true;pilotSubmit.textContent='Submitting...';

    let data=Object.fromEntries(new FormData(pilotForm));data.submission_id=submissionId();data.timestamp=new Date().toISOString();

    pilotLeads.push(data);localStorage.setItem('kidovaPilotLeads',JSON.stringify(pilotLeads));

    const sent=await sendOrQueue({type:'pilot',data});

    pilotStatus.classList.remove('hidden');pilotStatus.innerHTML=sent?'<b>Pilot interest submitted.</b> Thank you!':'<b>Saved on this device.</b> It will retry when the internet connection returns.';

    pilotForm.reset();pilotSubmit.disabled=false;pilotSubmit.textContent='Submit pilot interest';

  });

  function downloadCSV(rows,name){if(!rows.length)return alert('No records yet.');let keys=[...new Set(rows.flatMap(Object.keys))],cell=v=>'"'+String(v??'').replaceAll('"','""')+'"',csv=[keys.map(cell).join(','),...rows.map(r=>keys.map(k=>cell(r[k])).join(','))].join('\n'),a=document.createElement('a');a.href=URL.createObjectURL(new Blob([csv],{type:'text/csv'}));a.download=name;a.click()}

  function exportCSV(){downloadCSV(responses,'kidova-anonymous-survey.csv')}

  function exportPilots(){downloadCSV(pilotLeads,'kidova-pilot-contacts.csv')}

  window.addEventListener('online',retryPending);retryPending();

  if(location.hash&&document.querySelector(location.hash+'.screen'))show(location.hash.slice(1));
